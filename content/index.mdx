```python
from emunium import Browser, expect

# Launches installed Chrome via private pipe (no TCP debugging port)
with Browser(browser="chrome", motion="natural") as browser:
    browser.goto("https://app.example.com/login")

    browser.get_by_label("Email").type("me@example.com")
    browser.get_by_placeholder("Password").type("secret")
    browser.get_by_role("button", name="Sign in").click()

    expect(browser).to_have_url("*/dashboard")
```

<div className="engine-code engine-code-gecko">

```python
from emunium import Browser, expect

# Connects via signed policy extension (zero WebDriver flags)
with Browser.connect(
    browser=r"C:\Camoufox\camoufox.exe",
    user_data_dir=r"C:\Camoufox\profiles\work",
) as browser:
    browser.goto("https://bot.sannysoft.com/")

    expect(browser).to_have_url("https://bot.sannysoft.com/")
```
